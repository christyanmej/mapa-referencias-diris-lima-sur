ESCRITURA PROTEGIDA EN RANGOS EDITABLES (doPost) function doPost(e) { try { let payload; if (e.postData && e.postData.contents) { payload = JSON.parse(e.postData.contents); } else { payload = e.parameter; } // Validar clave de autorización if (String(payload.clave).trim() !== "6c07") { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: "Clave de autorización incorrecta." })) .setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("EESS_CAMAS") || ss.getSheetByName("EESS_CAMAS_2026") || ss.getSheets()[0]; const lastCol = sheet.getLastColumn(); const data = sheet.getDataRange().getDisplayValues(); const headers = data[0].map(h => String(h).trim()); // Identificar columna 'cod_ren' o 'Id' para buscar la fila let renColumnIndex = headers.indexOf("cod_ren"); if (renColumnIndex === -1) renColumnIndex = headers.indexOf("Id"); if (renColumnIndex === -1) throw new Error("No se encontró la columna 'cod_ren' en la hoja."); let rowIndexToUpdate = -1; const targetCodRen = String(payload.cod_ren || payload.Id || "").trim(); for (let i = 1; i < data.length; i++) { let cellValue = String(data[i][renColumnIndex]).trim(); if (cellValue === targetCodRen || parseInt(cellValue, 10) === parseInt(targetCodRen, 10)) { rowIndexToUpdate = i + 1; // Fila real en la hoja (1-based) break; } } if (rowIndexToUpdate > 0) { // Preprocesamiento numérico const camasOcup = parseInt(payload.camas_ocupadas) || 0; const camasDisp = parseInt(payload.camas_disponibles) || 0; const camillasOcup = parseInt(payload.camillas_ocupadas) || 0; const camillasDisp = parseInt(payload.camillas_disponibles) || 0; const camasTotales = camasOcup + camasDisp; const porcOcupCamas = camasTotales > 0 ? (camasOcup / camasTotales) : 0; const camillasTotales = camillasOcup + camillasDisp; const porcOcupCamillas = camillasTotales > 0 ? (camillasOcup / camillasTotales) : 0; const valObs = payload.observación || payload.observacion || ""; // Mapa de valores para las columnas F a AB const valoresEditables = { "fecha": payload.fecha || Utilities.formatDate(new Date(), "GMT-5", "dd/MM/yy"), "tipo_atencion": payload.tipo_atencion || "", "estado_uviclin": payload.estado_uviclin || "", "camas_ocupadas": camasOcup, "camas_disponibles": camasDisp, "camas_totales": camasTotales, "porc_ocupacion": porcOcupCamas, "camas_mosquiteros": parseInt(payload.camas_mosquiteros) || 0, "camillas_ocupadas": camillasOcup, "camillas_disponibles": camillasDisp, "camillas_totales": camillasTotales, "porc_ocup_camill": porcOcupCamillas, "camillas_mosquiteros": parseInt(payload.camillas_mosquiteros) || 0, "espacio_camas": payload.espacio_camas || "", "observación": valObs, "observacion": valObs, "brigadas": payload.brigadas || "", "rrhh": payload.rrhh || "", "horario_uvi": payload.horario_uvi || "", "telefono_eess": payload.telefono_eess || "", "jefe_eess": payload.jefe_eess || "", "telefono_jefe_eess": payload.telefono_jefe_eess || "", "dias_atencion": payload.dias_atencion || "", "correo_eess": payload.correo_eess || "" }; // Recorremos ÚNICAMENTE desde la Columna F (6) hasta la Columna AB (28) // Esto protege al 100% las Columnas A-E y AC-AF de ser modificadas for (let col = 6; col <= 28; col++) { let headerName = String(headers[col - 1] || "").trim(); // Buscar coincidencia en nuestro mapa (con o sin tilde para observación) if (valoresEditables.hasOwnProperty(headerName)) { sheet.getRange(rowIndexToUpdate, col).setValue(valoresEditables[headerName]); } } return ContentService .createTextOutput(JSON.stringify({ status: "success", message: "Establecimiento actualizado correctamente" })) .setMimeType(ContentService.MimeType.JSON); } else { throw new Error("No se encontró el código RENAES (" + targetCodRen + ") en la hoja."); } } catch (error) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
